\begin{algorithm}[htbp]
\caption{AdaptiveFedAvg Aggregation}
\label{alg:adaptivefedavg}
\begin{algorithmic}[1]
\REQUIRE Set of participating clients $\mathcal{K}$, Global model $w_t$, Round $t$
\ENSURE Updated global model $w_{t+1}$
\FOR{each client $k \in \mathcal{K}$ \textbf{in parallel}}
    \STATE $w_{t+1}^k \leftarrow \text{ClientUpdate}(k, w_t)$
    \STATE Compute local validation loss $L_k$ and dataset size $n_k$
    \STATE Send $w_{t+1}^k, L_k, n_k$ to aggregation server
\ENDFOR
\STATE Server receives updates from all $k \in \mathcal{K}$
\FOR{each client $k \in \mathcal{K}$}
    \STATE Compute quality score $q_k = e^{-L_k}$
    \STATE Compute rarity bonus $\rho_k = \frac{1}{\sqrt{F_{normal}^k}}$
    \STATE Compute reliability gate $g_k = 1 - \alpha \cdot \max(0, V_k - V_{threshold})$
    \STATE Calculate client weight: $\omega_k = n_k \cdot q_k \cdot \rho_k \cdot g_k$
\ENDFOR
\STATE Normalize weights: $\hat{\omega}_k = \frac{\omega_k}{\sum_{j \in \mathcal{K}} \omega_j}$
\STATE Aggregate: $w_{t+1} \leftarrow \sum_{k \in \mathcal{K}} \hat{\omega}_k w_{t+1}^k$
\RETURN $w_{t+1}$
\end{algorithmic}
\end{algorithm}
